Create `context/providers/trades-slice-provider.tsx`:
```typescript
'use client'
import { createContext, useContext, useState, useCallback } from 'react'
import type { Trade } from '@pris`ma/client'

interface TradesSliceContext {
  trades: Trade[]
  setTrades: (trades: Trade[]) => void
  addTrade: (trade: Trade) => void
  removeTrade: (id: string) => void
  updateTrade: (id: string, updates: Partial<Trade>) => void
}

const TradesSliceContext = createContext<TradesSliceContext>({
  trades: [],
  setTrades: () => {},
  addTrade: () => {},
  removeTrade: () => {},
  updateTrade: () => {},
})

export function TradesSliceProvider({ children, initialTrades = [] }: { children: React.ReactNode; initialTrades?: Trade[] }) {
  const [trades, setTrades] = useState<Trade[]>(initialTrades)
  
  const addTrade = useCallback((trade: Trade) => {
    setTrades(prev => [trade, ...prev])
  }, [])
  
  const removeTrade = useCallback((id: string) => {
    setTrades(prev => prev.filter(t => t.id !== id))
  }, [])
  
  const updateTrade = useCallback((id: string, updates: Partial<Trade>) => {
    setTrades(prev => prev.map(t => t.id === id ? { ...t, ...updates } : t))
  }, [])
  
  return (
    <TradesSliceContext.Provider value={{ trades, setTrades, addTrade, removeTrade, updateTrade }}>
      {children}
    </TradesSliceContext.Provider>
  )
}

export function useTradesSlice() {
  return useContext(TradesSliceContext)
}
```

Create similar slices for `accounts-slice-provider.tsx`, `filters-slice-provider.tsx`, and `derived-slice-provider.tsx`.
