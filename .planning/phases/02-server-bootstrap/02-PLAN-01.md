Create `server/dashboard-bootstrap.ts`:

```typescript
import { cache } from 'next/cache'
import { getDatabaseUserId } from 'lib/auth'
import { getUserData } from 'server/user-data'
import { getTradesAction } from 'server/trades'
import { calculateStatistics } from 'lib/statistics'
import { deriveScoreMetricsFromTrades } from 'lib/score-calculator'
import type { DashboardBootstrapPayload } from 'lib/types/bootstrap'

const PAGE_SIZE = 500

export async function getDashboardBootstrap(): Promise<DashboardBootstrapPayload> {
  const userId = await getDatabaseUserId()
  if (!userId) {
    return createEmptyBootstrap()
  }

  // Load core user data
  const userData = await getUserData(userId)
  const { user, subscription, dashboardLayout, accounts, groups, tags } = userData

  // Load first page of trades
  const tradesResult = await getTradesAction({ page: 1, pageSize: PAGE_SIZE })

  // Precompute analytics on raw trades
  const statistics = calculateStatistics(tradesResult.trades, accounts)
  const scoreMetrics = deriveScoreMetricsFromTrades(tradesResult.trades)

  return {
    user: user ? { id: user.id, email: user.email, fullName: user.fullName, avatarUrl: user.avatarUrl } : null,
    subscription: subscription ? { id: subscription.id, plan: subscription.plan, status: subscription.status, expiresAt: subscription.expiresAt } : null,
    dashboardLayout,
    timezone: userData.timezone || 'America/New_York',
    isAdmin: userData.isAdmin || false,
    accounts,
    groups,
    tags,
    trades: tradesResult.trades,
    tradesPagination: {
      page: 1,
      pageSize: PAGE_SIZE,
      totalCount: tradesResult.totalCount,
      hasMore: tradesResult.trades.length === PAGE_SIZE,
    },
    statistics,
    scoreMetrics,
    tickDetails: userData.tickDetails || [],
    financialEvents: userData.financialEvents || [],
    bootstrappedAt: new Date().toISOString(),
  }
}

function createEmptyBootstrap(): DashboardBootstrapPayload {
  return {
    user: null,
    subscription: null,
    dashboardLayout: null,
    timezone: 'America/New_York',
    isAdmin: false,
    accounts: [],
    groups: [],
    tags: [],
    trades: [],
    tradesPagination: { page: 1, pageSize: PAGE_SIZE, totalCount: 0, hasMore: false },
    statistics: createEmptyStatistics(),
    scoreMetrics: createEmptyScoreMetrics(),
    tickDetails: [],
    financialEvents: [],
    bootstrappedAt: new Date().toISOString(),
  }
}

function createEmptyStatistics(): import('lib/statistics').StatisticsProps {
  return { totalPnL: 0, winRate: 0, totalTrades: 0, avgWin: 0, avgLoss: 0, profitFactor: 0, largestWin: 0, largestLoss: 0 }
}

function createEmptyScoreMetrics(): import('lib/score-calculator').ScoreMetrics {
  return { score: 0, grade: 'N', factors: {} }
}
```

Wrap with cache. Use `'use cache'` if the project supports it, otherwise use `unstable_cache`:
```typescript
// At top of file, after imports
export const getDashboardBootstrapCached = cache(async () => {
  'use cache'
  cacheLife({ stale: 60, revalidate: 60, expire: 300 })
  return getDashboardBootstrap()
}, ['dashboard-bootstrap'])
```

```typescript
export function shouldUseServerBootstrap(userId?: string): boolean {
  if (!FEATURE_FLAGS.SERVER_DASHBOARD_BOOTSTRAP) return false
  const pct = FEATURE_FLAGS.SERVER_DASHBOARD_ROLLOUT_PCT
  if (pct >= 100) return true
  if (pct <= 0) return false
  if (userId) {
    const hash = userId.split('').reduce((acc, c) => ((acc << 5) - acc + c.charCodeAt(0)) | 0, 0)
    return Math.abs(hash) % 100 < pct
  }
  return false
}
```
